Google 試算表 (Google Sheets) 雙向雲端記帳綁定

免綁信用卡、完全免費！1 分鐘將您的 Google 試算表變成雙人手機同步帳本

🛠️ 1 分鐘設定步驟（只需設定一次）：
步驟 1：建立空白 Google 試算表並開啟 Apps Script

1. 點擊前往 https://sheets.new 建立一份空白試算表（可命名為「2027京都自由行記帳」）。

2. 點選試算表頂端選單的 「擴充功能 (Extensions)」→「Apps Script」。

步驟 2：一鍵複製下方程式碼並貼入 Apps Script

將 Apps Script 編輯器裡原本的 function myFunction() {} 刪掉，貼上下方程式碼並按 Ctrl+S（或磁片圖示）儲存：

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('京都記帳') || ss.insertSheet('京都記帳');
  var data = sheet.getDataRange().getValues();
  var expenses = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0] || row[0] === 'SUMMARY') continue;
    expenses.push({
      id: String(row[0]),
      day: String(row[1]),
      category: String(row[2]),
      title: String(row[3]),
      currency: String(row[4]),
      amount: Number(row[5]),
      payer: String(row[7]),
      split: String(row[8])
    });
  }
  var metaSheet = ss.getSheetByName('設定');
  var meta = {};
  if (metaSheet && metaSheet.getLastRow() >= 2) {
    var m = metaSheet.getRange(2, 1, 1, 3).getValues()[0];
    meta = { A: m[0], B: m[1], rate: Number(m[2]) };
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', expenses: expenses, meta: meta }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('京都記帳') || ss.insertSheet('京都記帳');
  sheet.clearContents();
  sheet.appendRow(['ID', '所屬日期', '支出分類', '項目名稱', '幣別', '原始金額', '折合台幣(NT$)', '墊付代碼', '分攤代碼', '墊付人顯示', '分攤說明', '同步時間']);
  var rate = Number(payload.rate) || 0.215;
  var names = payload.names || { A: '旅伴 A', B: '旅伴 B' };
  var list = payload.expenses || [];
  var nowStr = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy-MM-dd HH:mm');
  var coreTwd = 0;
  var shopTwd = 0;
  for (var i = 0; i < list.length; i++) {
    var it = list[i];
    var twd = it.currency === 'JPY' ? Math.round(Number(it.amount) * rate) : Math.round(Number(it.amount));
    if (it.category === '購物伴手禮') shopTwd += twd; else coreTwd += twd;
    var pLabel = it.payer === 'A' ? names.A : (it.payer === 'B' ? names.B : '公費錢包');
    var sLabel = it.split === 'EVEN' ? '兩人平分' : (it.split === 'SELF' ? '個人自付' : '代墊對方');
    sheet.appendRow([it.id, it.day, it.category, it.title, it.currency, it.amount, twd, it.payer, it.split, pLabel, sLabel, nowStr]);
  }
  sheet.appendRow(['SUMMARY', '', '核心旅費合計(不含購物)', '', 'TWD', '', coreTwd, '', '', '購物合計 NT$' + shopTwd, '總花費 NT$' + (coreTwd + shopTwd), nowStr]);
  var metaSheet = ss.getSheetByName('設定') || ss.insertSheet('設定');
  metaSheet.clearContents();
  metaSheet.appendRow(['旅伴A姓名', '旅伴B姓名', '換算匯率']);
  metaSheet.appendRow([names.A, names.B, rate]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: list.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
步驟 3：部署為「網頁應用程式 (Web app)」

1. 點擊右上角藍色按鈕 「部署 (Deploy)」→「新增部署作業 (New deployment)」。

2. 點選左上角齒輪圖示 → 選擇 「網頁應用程式 (Web app)」。

3. 「執行身分」選 「我 (Me)」；「誰可以存取 (Who has access)」請務必選擇「所有人 (Anyone)」。

4. 點擊「部署」並完成授權（若跳出「Google 尚未驗證」，點左下角「進階」→「前往專案 (不安全)」→「允許」）。

5. 複製產生的 「網頁應用程式網址 (`https://script.google.com/macros/s/.../exec`)」 貼回上方輸入框即完成！
